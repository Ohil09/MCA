SET SERVEROUTPUT ON
DECLARE
v_patient_ID patient.patient_ID%TYPE;
v_patient_name patient.patient_name%TYPE;
v_dob patient.DOB%TYPE;

BEGIN
	v_patient_ID := '&patient_ID';

	SELECT patient_name,DOB INTO v_patient_name,v_dob FROM patient WHERE patient_ID = v_patient_ID;
	DBMS_OUTPUT.PUT_LINE('Patient Name : ' || v_patient_name);
	DBMS_OUTPUT.PUT_LINE('Patient DOB : ' || v_dob);
EXCEPTION
	WHEN NO_DATA_FOUND THEN
		DBMS_OUTPUT.PUT_LINE('Error: No such patient Exists !');
	WHEN TOO_MANY_ROWS THEN
		DBMS_OUTPUT.PUT_LINE('Error : More than one record is retrieved !');
	WHEN OTHERS THEN 
		DBMS_OUTPUT.PUT_LINE('Error : Some other Error Occurred !');
END;
/
