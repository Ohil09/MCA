
CREATE TABLE BILL(
    Bill_ID number(6) CONSTRAINT bill_pk Primary Key,
    Appointment_ID NUMBER(6) CONSTRAINT bill_appt_fk REFERENCES Appointment(Appointment_ID),
    Bill_Date DATE DEFAULT SYSDATE NOT NULL,
    Consultation_amt NUMBER(8,2) CONSTRAINT const_chk check (Consultation_amt > 0),
    Discount_per NUMBER(8,2) CONSTRAINT disct_chk check (Discount_per BETWEEN 0 AND 30),
    Total_amt NUMBER(8,2) CONSTRAINT total_chk check (Total_amt > 0),
    Payment_status varchar(10) CONSTRAINT bill_status_chk check(Payment_status IN ('PENDING','PARTIAL','PAID')),
        CONSTRAINT one_bill_chk UNIQUE(Appointment_ID));
