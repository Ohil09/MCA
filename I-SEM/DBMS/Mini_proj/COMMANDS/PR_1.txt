CREATE OR REPLACE PROCEDURE record_sale (
    p_sale_id IN NUMBER,
    p_toy_id IN NUMBER,
    p_sale_date IN DATE,
    p_quantity IN NUMBER,
    p_unit_price IN NUMBER
)
IS
    v_stock TOY.current_stock%TYPE;
BEGIN
    SELECT current_stock
    INTO v_stock
    FROM TOY
    WHERE toy_id = p_toy_id
    FOR UPDATE;

    IF p_quantity <= 0 OR p_quantity > v_stock THEN
        RAISE_APPLICATION_ERROR(-20001, 'Invalid quantity or insufficient stock');
    END IF;

    INSERT INTO SALE (
        sale_id, toy_id, sale_date, quantity_sold, unit_price
    )
    VALUES (
        p_sale_id, p_toy_id, p_sale_date, p_quantity, p_unit_price
    );

    UPDATE TOY
    SET current_stock = current_stock - p_quantity
    WHERE toy_id = p_toy_id;
END;
