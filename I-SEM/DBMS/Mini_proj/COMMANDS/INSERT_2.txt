-- 9. Verify stock and sales for the four newly added toys
SELECT t.toy_id,
       t.toy_name,
       t.current_stock,
       NVL(SUM(s.quantity_sold), 0) AS total_quantity_sold
FROM TOY t
LEFT JOIN SALE s ON t.toy_id = s.toy_id
WHERE t.toy_id BETWEEN 209 AND 212
GROUP BY t.toy_id, t.toy_name, t.current_stock
ORDER BY t.toy_id;
