SET SERVEROUTPUT ON
DECLARE
	v_eno emp.empno%TYPE;
	v_ename emp.ename%TYPE;
	v_dept dept.dname%TYPE;
	v_sal emp.sal%TYPE;
BEGIN
	v_eno := &emp_no;
	SELECT e.ename,d.dname,e.sal INTO v_ename,v_dept,v_sal FROM emp e JOIN dept d ON e.deptno = d.dno WHERE e.empno = v_eno;
	DBMS_OUTPUT.PUT_LINE(v_ename||' works in '||v_dept||' Department and draws '||v_sal||'/- as salary');

EXCEPTION
	WHEN NO_DATA_FOUND THEN
		DBMS_OUTPUT.PUT_LINE('Please Enter an valid Emp no that has an Dept number!');
END;
/
