CREATE OR REPLACE PROCEDURE CALC_YEARS (
    P_EMP        IN EMP.EMPNO%TYPE,
    P_AGE        OUT NUMBER,
    P_EXPERIENCE OUT NUMBER
)
IS
    V_DOB EMP.DATE_BIRTH%TYPE;
    V_DOJ EMP.DATE_JOIN%TYPE;
BEGIN
    SELECT DATE_BIRTH, DATE_JOIN
    INTO V_DOB, V_DOJ
    FROM EMP
    WHERE EMPNO = P_EMP;

    P_AGE := TRUNC(MONTHS_BETWEEN(SYSDATE, V_DOB) / 12);
    P_EXPERIENCE := TRUNC(MONTHS_BETWEEN(SYSDATE, V_DOJ) / 12);
END;
