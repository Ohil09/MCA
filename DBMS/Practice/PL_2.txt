CREATE OR REPLACE PROCEDURE PROCESS_PAYMENT (
    p_bill_id IN NUMBER,
    p_amount  IN NUMBER
)
IS
    v_bill_amount BILL.TOTAL_AMOUNT%TYPE;
    v_total_paid  NUMBER(10,2);
    v_new_total   NUMBER(10,2);
    v_status      BILL.PAYMENT_STATUS%TYPE;
BEGIN
    -- Validate payment amount
    IF p_amount IS NULL OR p_amount <= 0 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'Payment amount must be greater than zero.'
        );
    END IF;

    -- Check whether bill exists and lock it
    SELECT TOTAL_AMOUNT
    INTO v_bill_amount
    FROM BILL
    WHERE BILL_ID = p_bill_id
    FOR UPDATE;

    -- Calculate previous payments
    SELECT NVL(SUM(PAYMENT_AMOUNT), 0)
    INTO v_total_paid
    FROM PAYMENT
    WHERE BILL_ID = p_bill_id;

    v_new_total := v_total_paid + p_amount;

    -- Prevent overpayment
    IF v_new_total > v_bill_amount THEN
        RAISE_APPLICATION_ERROR(
            -20002,
            'Payment exceeds the outstanding balance.'
        );
    END IF;

    -- Insert payment record
    INSERT INTO PAYMENT (
        PAYMENT_ID,
        BILL_ID,
        PAYMENT_DATE,
        PAYMENT_AMOUNT
    )
    VALUES (
        payment_seq.NEXTVAL,
        p_bill_id,
        SYSDATE,
        p_amount
    );

    -- Determine payment status
    IF v_new_total = 0 THEN
        v_status := 'PENDING';
    ELSIF v_new_total < v_bill_amount THEN
        v_status := 'PARTIAL';
    ELSE
        v_status := 'PAID';
    END IF;

    -- Update bill status
    UPDATE BILL
    SET PAYMENT_STATUS = v_status
    WHERE BILL_ID = p_bill_id;

    DBMS_OUTPUT.PUT_LINE('Payment recorded successfully.');
    DBMS_OUTPUT.PUT_LINE('Total Paid: ' || v_new_total);
    DBMS_OUTPUT.PUT_LINE('Payment Status: ' || v_status);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(
            -20003,
            'Invalid Bill ID.'
        );
END;
